      my $forwardEnd = $finnerInfo->getLocation() + $finnerInfo->getLength() - 1;

      my $maxReverseLocation = $forwardStart + $signatureMaxLength - 1;

      my $previousCompatibleIndexFound = $FALSE;

            next;
          }

            $previousFirstCompatibleIndex = $j;

          }

          last;
